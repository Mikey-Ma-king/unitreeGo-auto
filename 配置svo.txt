3. launch文件修改
sudo chown -R unitree:unitree /unitree/module/Odometer_service/src/rpg_svo_pro_open/svo_ros/launch/

3.1 添加bridge.yaml 
topics:
  - 
    topic: /svo/pose_cam_0 # ROS1 topic name
    type: geometry_msgs/msg/PoseStamped  # ROS2 type name (assuming the message type is Image)
    queue_size: 1  # For the publisher back to ROS1
  - 
    topic: /camera/infra1/image_rect_raw  # ROS1 topic name
    type: sensor_msgs/msg/Image  # ROS2 type name (assuming the message type is Image)
    queue_size: 1  # For the publisher back to ROS1


3.2 修改rs_camera.launch
<launch>

  <node pkg="svo_ros" type="svo_node" name="svo" respawn="true" clear_params="true" output="screen">

  <remap from="/svo/image/0" to="/svo/image_0" />
  <remap from="/svo/pose_cam/0" to="/svo/pose_cam_0" />
  
  <!-- Camera topic to subscribe to -->
    <param name="cam0_topic" value="/camera/infra1/image_rect_raw" type="str" />

  <!-- Camera calibration file -->
    <param name="calib_file" value="$(find svo_ros)/param/calib/rs_camera_calib.yaml" type="str" />

  <!-- Parameters -->
    <rosparam file="$(find svo_ros)/param/rs_camera_param.yaml" />

  <!-- Initial camera orientation, make it point downwards -->
    <param name="init_rx" value="3.14" />
    <param name="init_ry" value="0.00" />
    <param name="init_rz" value="0.00" />
  </node>

</launch>
